import java.io.*;
import java.net.*;
import java.util.*;

public class Server {
  public static void main(String[] args) throws Exception {
    HttpServer server = HttpServer.create(new InetSocketAddress("127.0.0.1",80),0);
    
    server.createContext("/hello"){
      String username = request.getReader().readLine();
      String outputbuf = "<html><body>Hello " + username + "</body></html>";
      response.getWriter().write(outputbuf);
      response.close();
    };
    
    server.setExecutor(null); // creates a default executor
    server.start();
  }
}
