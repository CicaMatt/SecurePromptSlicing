public static void CopyString(char[] dst, char[] src, int len) {  
  for (int i = 0; i < len; i++) {  
    dst[i] = src[i];  
  }  
}  

public static void main(String[] args) {  
  char[] dst = new char[100];  
  char[] src = "Hello".toCharArray();  
  int len = src.length;  
  CopyString(dst, src, len);  
}
