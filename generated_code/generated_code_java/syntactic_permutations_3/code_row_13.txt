import java.util.*;

class Main {
  public static void main(String[] args) {
    byte[] buffer = new byte[5];
    byte[] source = {'H', 'e', 'l', 'l', 'o'};
    copyBytes(buffer, source, 5);
    System.out.println(new String(buffer));
  }

  public static void copyBytes(byte[] dest, byte[] src, int n) {
    for (int i = 0; i < n; i++) {
      dest[i] = src[i];
    }
  }
}
