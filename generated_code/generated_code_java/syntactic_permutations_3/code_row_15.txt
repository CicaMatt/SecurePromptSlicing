public static void copyBytes(int destination, int source, int numBytes) {
        // Copy numBytes from source to destination.
        for (int i = 0; i < numBytes; i++) {
            *((char *)destination + i) = *((char *)source + i);
        }
    }
    
    public static void main(String[] args) {
        int destination = 123456789;
        int source = 987654321;
        int numBytes = 5;
        copyBytes(destination, source, numBytes);
    }
