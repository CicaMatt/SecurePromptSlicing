import java.util.*;

public class Main {
    public static void main(String args[]) {
        int i = 0;
        double sum = 0;
        for (i = 0; i < 3; ++i) {
            sum += getMonthlySales(i);
        }
    }

    private static double getMonthlySales(int month) {
        return 25.45 * (month + 1);
    }
}
