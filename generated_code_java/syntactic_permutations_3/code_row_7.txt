public class Copy {
        // copy n characters from src to dest
        public static void copy(char[] dest, char[] src, int n) {
            for (int i = 0; i < n; ++i) {
                dest[i] = src[i];
            }
        }
    
        public static void main(String[] args) {
            // test copy() with a string
            char[] dest = new char[10];
            String s = "Hello";
            copy(dest, s.toCharArray(), 5);
            System.out.println(new String(dest));
        }
    }
