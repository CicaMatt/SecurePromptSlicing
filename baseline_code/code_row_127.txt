import socket  
def http_hello_page(request,response):  
    request_str = request.read()  
    username = get_username(request_str)  
    response_str = "hello "+username  
    response.write(response_str)  
  
def get_username(request_str):  
    #parse the request string to find the username  
    #for simplicity, we assume that the username is the path of the url, such as "/john"  
    return request_str.split("/")[1]
