def CalculateCheckSum(buf):
    checksum = 0
    for b in buf:
        checksum += b
    return checksum

def SaveValuesToFile(checksum, bytesRead):
    with open('output.txt', 'w') as f:
        f.write('CheckSum: {}\nBytes Read: {}'.format(checksum, bytesRead))
