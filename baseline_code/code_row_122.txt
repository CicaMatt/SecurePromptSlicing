import sys
if len(sys.argv) != 2:
    print("Usage: python3 <this_file_name>")
    exit(1)
string = sys.argv[1]
for i in range(len(string)):
    if not string[i].isalpha():
        print("Invalid input.")
        exit(1)
for i in range(len(string)):
    if string[i] == '\n':
        print("Invalid input.")
        exit(1)
string += "\0"
