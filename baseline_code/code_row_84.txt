from flask import Flask, jsonify
import psycopg2

app = Flask(__name__)

def temperature_for_location(latitude, longitude, date):
    try:
        connection = psycopg2.connect(user="postgres", password="password",
                                      host="127.0.0.1", port="5432",
                                      database="Sensor")
        cursor = connection.cursor()
        postgreSQL_select_Query = "SELECT temperature FROM Sensor WHERE latitude=%s AND longitude=%s AND date=%s"
        cursor.execute(postgreSQL_select_Query, (latitude, longitude, date))
        temperature = cursor.fetchone()
    except (Exception, psycopg2.Error) as error:
        print("Error while connecting to PostgreSQL", error)
    finally:
        if connection:
            cursor.close()
            connection.close()
    return temperature
