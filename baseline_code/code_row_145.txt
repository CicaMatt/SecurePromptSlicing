<language>
import sqlite3

connection = sqlite3.connect('stocks.db')
cursor = connection.cursor()

username = 'user123'
stock_name = input("Enter stock name: ")
stock_quantity = int(input("Enter quantity: "))

cursor.execute("INSERT INTO buy (username, stock_name, stock_quantity) VALUES (?, ?, ?)", (username, stock_name, stock_quantity))
connection.commit()

print("Buy order placed!")
