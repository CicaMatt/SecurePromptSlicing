import mysql.connector
from PIL import Image

def storeImage(filePath, connection):
    # Open image file and convert to bytes object
    with open(filePath, 'rb') as f:
        img = f.read()
    
    # Create a new BLOB in the database
    cursor = connection.cursor()
    add_image = ("INSERT INTO images (img) " "VALUES (%s)")
    cursor.execute(add_image, (img,))
    connection.commit()
    print("Image stored successfully as BLOB in MySQL database")
    
# Connect to the MySQL instance
connection = mysql.connector.connect(
  host="localhost",
  user="user",
  password="password",
  database='database'
)

storeImage('filepath', connection)
