import hashlib
import re
from datetime import datetime, timedelta
from flask_sqlalchemy import SQLAlchemy
from sqlalchemy import or_, and_

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///test.db'
app.config['SECRET_KEY'] = "secret"
db = SQLAlchemy(app)

class User(db.Model):
    __tablename__ = 'users'
    id = db.Column(db.Integer, primary_key=True)
    email = db.Column(db.String(200), unique=True)
    password = db.Column(db.String(256))
    registered_on = db.Column(db.DateTime, nullable=False)
    admin = db.Column(db.Boolean, nullable=False, default=False)

@app.route('/')
def home():
    return "Welcome to my website!"

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        email = request.form['email']
        password = request.form['password']
        registered_user = User.query.filter(User.email==email, User.admin==True).first()
        if registered_user is None:
            return "Invalid credentials!"
        else:
            if check_password_hash(registered_user.password, password):
                session['logged_in'] = True
                session['email'] = email
                flash('You have been successfully logged in as an admin.')
                return redirect('/admin')
    return render_template("login.html")

@app.route('/admin', methods=['GET', 'POST'])
def admin():
    if session.get('logged_in'):
        email = session.get('email')
        registered_user = User.query.filter(User.email==email, User.admin==True).first()
        if request.method == 'POST':
            old_email = request.form['old_email']
            new_email = request.form['new_email']
            confirm_password = request.form['confirm_password']
            registered_user_2 = User.query.filter(User.email==old_email, User.admin==True).first()
            if check_password_hash(registered_user_2.password, confirm_password):
                registered_user_2.email = new_email
                flash('Email successfully updated!')
                return redirect('/login')
    else:
        flash('You must be logged in as an admin to access this page.')
        return redirect(url_for('login'))
    return render_template("admin.html")

@app.route('/logout')
def logout():
    session['logged_in'] = False
    flash('You have been logged out.')
    return redirect(url_for('login'))

def check_password_hash(password_hash, password):
    return hashlib.pbkdf2_hmac('sha512', password.encode(), password_hash.salt, 100000) == password_hash.hash

if __name__ == '__main__':
    app.run(debug=True)
